```python
"""What a room is, for a caller that narrates it.

`graph` knows connectivity and `play` knows rendering; this module answers
"what is this room, and what may the players know about it". Every consumer
of room text — the backend's campaign contract, its sessions routes, the MCP
server — goes through here, so the rules for what is secret live in one place.

Output is split in two. `perceived` is what a character standing in the room
can see: boxed text, visible features, the ways out it has found. `dm_only`
is the rest: DM notes, unfound secret doors, traps, secret features. A caller
serving players drops `dm_only`; a caller serving the DM keeps both, so the DM
knows what not to say.
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Iterable, Mapping, Optional

from .geometry import _point_strictly_inside, corridor_polygons, room_polygon
from .graph import Graph, is_blocked
from .model import Corridor, DungeonMap, Room


@dataclass(frozen=True)
class SessionView:
    """The runtime overlay a play session carries, detached from any ORM."""

    discovered_nodes: frozenset[str] = frozenset()
    discovered_doors: frozenset[str] = frozenset()
    door_states: Mapping[str, str] = field(default_factory=dict)
    party_location: Optional[str] = None

    def state(self, key: str, authored: str) -> str:
        return self.door_states.get(key, authored)


# ----- lookup -----

def _nodes(dmap: DungeonMap) -> dict[str, Room | Corridor]:
    out: dict[str, Room | Corridor] = {}
    for name, r in dmap.rooms.items():
        out[f"room.{name}"] = r
    for name, c in dmap.corridors.items():
        out[f"corridor.{name}"] = c
    for layer in dmap.layers:
        for r in layer.rooms:
            out.setdefault(f"room.{r.name}", r)
        for c in layer.corridors:
            out.setdefault(f"corridor.{c.name}", c)
    return out


def node_label(dmap: DungeonMap, node_id: str) -> str:
    """Label text, else a corridor's display name, else the bare name."""
    obj = _nodes(dmap).get(node_id)
    bare = node_id.split(".", 1)[-1]
    if obj is None:
        return bare
    if obj.label is not None and obj.label.text:
        return obj.label.text
    display = getattr(obj, "display_name", None)
    return display or bare


def candidates(dmap: DungeonMap, graph: Graph, query: str) -> list[dict]:
    """Every node `query` could mean: exact id, bare name, label or display
    name, case-insensitive. An exact id or bare-name match wins outright."""
    q = (query or "").strip()
    if not q:
        return []
    if graph.has_node(q):
        return [{"id": q, "label": node_label(dmap, q)}]
    low = q.lower()
    bare = [nid for nid in graph.nodes if nid.split(".", 1)[-1].lower() == low]
    if len(bare) == 1:
        return [{"id": bare[0], "label": node_label(dmap, bare[0])}]
    hits = []
    for nid, obj in _nodes(dmap).items():
        if nid not in graph.nodes:
            continue
        names = {nid.split(".", 1)[-1].lower()}
        if obj.label is not None and obj.label.text:
            names.add(obj.label.text.strip().lower())
        if getattr(obj, "display_name", None):
            names.add(obj.display_name.strip().lower())
        if low in names:
            hits.append({"id": nid, "label": node_label(dmap, nid)})
    return sorted(hits, key=lambda h: h["id"])


def resolve_node(dmap: DungeonMap, graph: Graph, query: str) -> Optional[str]:
    """The node id `query` names, or None if it names none or several."""
    hits = candidates(dmap, graph, query)
    return hits[0]["id"] if len(hits) == 1 else None


# ----- geometry: which node a loose point belongs to -----

def _area(poly: list) -> float:
    return abs(sum(poly[i][0] * poly[(i + 1) % len(poly)][1]
                   - poly[(i + 1) % len(poly)][0] * poly[i][1]
                   for i in range(len(poly)))) / 2.0


def _polys(nid: str, obj) -> list[list]:
    if nid.startswith("room."):
        return [room_polygon(obj)]
    return corridor_polygons(obj)


def _owner(dmap: DungeonMap, point) -> Optional[str]:
    """The smallest node whose outline contains `point` — caves drawn inside
    a canyon polygon own their own annotations."""
    best: tuple[float, str] | None = None
    for nid, obj in _nodes(dmap).items():
        for poly in _polys(nid, obj):
            if len(poly) >= 3 and _point_strictly_inside(point, poly):
                a = _area(poly)
                if best is None or a < best[0]:
                    best = (a, nid)
    return best[1] if best else None


# ----- exits -----

def node_exits(graph: Graph, node: str, view: SessionView, *,
               labels: Mapping[str, str]) -> tuple[list[dict], list[dict]]:
    """(perceived, secret) exits from `node`.

    Perceived: every non-concealed door and boundary opening, plus concealed
    ones the session has found, each with its effective state. Secret:
    concealed doors not yet found. `discovered` says whether the door itself
    has been seen, which callers use to split "known way out" from "the DM
    knows there is a door here the party hasn't noticed yet"."""
    perceived: list[dict] = []
    secret: list[dict] = []

    def entry(key, to, type_, authored):
        state = view.state(key, authored)
        return {
            "door": key,
            "to": to,
            "to_label": labels.get(to) if to else None,
            "type": type_,
            "state": state,
            "blocked": is_blocked(state),
            "discovered": key in view.discovered_doors,
            "far_side_explored": bool(to) and to in view.discovered_nodes,
        }

    for edge in graph.incident_edges(node):
        e = entry(edge.key, edge.other(node), edge.type, edge.state)
        if edge.hidden and edge.key not in view.discovered_doors:
            secret.append(e)
        else:
            perceived.append(e)
    for b in graph.boundary_exits(node):
        e = entry(b.key, None, b.type, b.state)
        if b.hidden and b.key not in view.discovered_doors:
            secret.append(e)
        else:
            perceived.append(e)
    return perceived, secret


# ----- the room -----

def _door_extras(dmap: DungeonMap) -> dict[str, object]:
    from .graph import door_key
    out = {}
    doors = list(dmap.doors)
    for layer in dmap.layers:
        doors.extend(layer.doors)
    for d in doors:
        out.setdefault(door_key(d), d)
    return out


def room_context(dmap: DungeonMap, graph: Graph, node: str, view: SessionView) -> dict:
    obj = _nodes(dmap)[node]
    labels = {nid: node_label(dmap, nid) for nid in graph.nodes}
    doors = _door_extras(dmap)
    secret_refs = {n for n, fd in dmap.feature_defs.items() if getattr(fd, "secret", False)}

    exits, secret_exits = node_exits(graph, node, view, labels=labels)
    for e in exits:
        d = doors.get(e["door"].split("#", 1)[0])
        if d is not None and d.description:
            e["description"] = d.description
    perceived_exits = [e for e in exits if e["discovered"]]
    undiscovered = [e for e in exits if not e["discovered"]]
    trapped = sorted(e["door"] for e in exits + secret_exits
                     if getattr(doors.get(e["door"].split("#", 1)[0]), "trapped", False))
    door_notes = [{"door": e["door"], "dm_notes": doors[e["door"].split("#", 1)[0]].dm_notes}
                  for e in exits + secret_exits
                  if getattr(doors.get(e["door"].split("#", 1)[0]), "dm_notes", None)]

    # Features: described ones individually, plain ones counted by kind.
    features: list[dict] = []
    counts: dict[str, int] = {}
    secret_features: list[dict] = []
    for f in obj.features:
        if f.secret or f.ref in secret_refs:
            item = {"name": f.ref}
            if f.description:
                item["description"] = f.description
            if f.dm_notes:
                item["dm_notes"] = f.dm_notes
            secret_features.append(item)
        elif f.description:
            features.append({"name": f.ref, "description": f.description})
        else:
            counts[f.ref] = counts.get(f.ref, 0) + 1
    features.extend({"name": n, "count": c} for n, c in sorted(counts.items()))
    feature_notes = [{"name": f.ref, "dm_notes": f.dm_notes} for f in obj.features
                     if f.dm_notes and not (f.secret or f.ref in secret_refs)]

    # Annotations: the node's own texts plus map-level texts inside it.
    texts = list(obj.texts) + [t for t in list(dmap.texts) + [t for l in dmap.layers for t in l.texts]
                               if _owner(dmap, t.position) == node]
    annotations = []
    annotation_notes = []
    for t in texts:
        a = {"text": t.text}
        if t.description:
            a["description"] = t.description
        annotations.append(a)
        if t.dm_notes:
            annotation_notes.append({"text": t.text, "dm_notes": t.dm_notes})

    # Cross-map exits: the node's own plus map-level ones inside it.
    xs = list(obj.exits) + [x for x in list(dmap.exits) + [x for l in dmap.layers for x in l.exits]
                            if _owner(dmap, x.position) == node]
    map_exits, secret_map_exits = [], []
    for x in xs:
        item = {"target_map": x.target_map,
                "label": x.label.text if x.label is not None else None}
        if x.description:
            item["description"] = x.description
        (secret_map_exits if x.secret else map_exits).append(item)

    return {
        "id": node,
        "kind": graph.nodes[node].kind,
        "label": labels[node],
        "discovered": node in view.discovered_nodes,
        "party_here": view.party_location == node,
        "perceived": {
            "description": obj.description,
            "features": features,
            "annotations": annotations,
            "exits": perceived_exits,
            "map_exits": map_exits,
        },
        "dm_only": {
            "notes": obj.dm_notes,
            "secret_exits": secret_exits,
            "undiscovered_exits": undiscovered,
            "trapped_doors": trapped,
            "door_notes": door_notes,
            "secret_features": secret_features,
            "feature_notes": feature_notes,
            "annotation_notes": annotation_notes,
            "secret_map_exits": secret_map_exits,
        },
    }


def known_map(dmap: DungeonMap, graph: Graph, view: SessionView) -> dict:
    """The discovered part of the map: nodes, connections between them, and
    the frontier — found doors leading somewhere not yet explored."""
    labels = {nid: node_label(dmap, nid) for nid in graph.nodes}
    nodes = [{"id": nid, "kind": graph.nodes[nid].kind, "label": labels[nid]}
             for nid in sorted(view.discovered_nodes) if nid in graph.nodes]
    connections, frontier, seen = [], [], set()
    for nid in sorted(view.discovered_nodes):
        if nid not in graph.nodes:
            continue
        for edge in graph.incident_edges(nid):
            if edge.key not in view.discovered_doors or edge.key in seen:
                continue
            seen.add(edge.key)
            state = view.state(edge.key, edge.state)
            a_in, b_in = edge.a in view.discovered_nodes, edge.b in view.discovered_nodes
            if a_in and b_in:
                connections.append({"door": edge.key, "between": [edge.a, edge.b],
                                    "type": edge.type, "state": state})
            else:
                known, unknown = (edge.a, edge.b) if a_in else (edge.b, edge.a)
                frontier.append({"door": edge.key, "from": known, "leads_to": unknown,
                                 "leads_to_label": labels[unknown],
                                 "type": edge.type, "state": state})
    return {"party_location": view.party_location, "nodes": nodes,
            "connections": connections, "frontier": frontier}
```

```python
    m = contract.map_in_link(db, external_id, map_id)

        hit = [e for e in graph.edges if {e.a, e.b} == {a, b}] if a and b else []

        if e.key == key:

    for bx in graph.boundary:
        if bx.key == key:
            authored = bx.state
    if key is None or authored is None:
        return _json_error(404, {"error": "unknown door", "door": body.door,
                                 "between": body.between})
    s = contract.campaign_session_for(db, m, external_id)
    if body.discovered:
        s.discovered_doors = sorted(set(s.discovered_doors or []) | {key})
    if body.state:
        states = dict(s.door_states or {})
        states[key] = body.state
        s.door_states = states
    db.commit()
    return {"ok": True, "door": key,
            "state": (s.door_states or {}).get(key, authored),
            "discovered": key in (s.discovered_doors or [])}
```

```python
            comp, stack = [], [start]

                comp.append(cur)

                    if cur in (e.a, e.b):

                        if nxt not in seen:

                            stack.append(nxt)
            parts.append(sorted(comp))

            diags.append(_diag("warning", f"map is not connected: {len(parts)} separate parts ({names})"))

                continue
            doors = {e.key for e in g.incident_edges(nid)} | {b.key for b in g.boundary_exits(nid)}
            doors |= {e.key for e in g.edges if e.b == nid and e.one_way}
```

```python
"""Where the party is.

One record, state.json["position"], answers it for every surface — the
primer, the /play sidebar, introduce_npc's auto-tag, move_party. It has two
halves, either of which may be empty: a ttrpg2 location (a town, a stretch of
road, a dungeon as a whole) and a node on a dungml map (a room).

Campaigns from before this record existed fall back once to the last
`location_visited` event; the first move writes the record.
"""
from __future__ import annotations

import _campaign as _c

_KEYS = ("location", "area", "name", "map_id", "map_name", "node", "label")


def _title(path) -> str:
    try:
        for line in path.read_text(encoding="utf-8").splitlines():
            if line.startswith("#"):
                return line.lstrip("#").strip()
    except OSError:
        pass
    return path.stem


def find_location(cfg: dict, slug: str) -> dict | None:
    """A location file by 'area/slug' or bare slug. None if there is none."""
    locs = cfg["_data_dir"] / "locations"
    slug = (slug or "").strip().strip("/")
    if not slug:
        return None
    direct = locs / f"{slug}.md"
    hits = [direct] if direct.exists() else sorted(locs.rglob(f"{slug.split('/')[-1]}.md"))
    if not hits:
        return None
    p = hits[0]
    area = "" if p.parent == locs else p.parent.relative_to(locs).as_posix()
    return {"slug": p.stem, "area": area, "name": _title(p), "path": p}


def current_position(cfg: dict, state: dict | None = None,
                     events: list | None = None) -> dict:
    state = state if state is not None else _c.load_state(cfg)
    pos = state.get("position")
    if isinstance(pos, dict) and pos:
        return dict(pos)
    for e in reversed(events if events is not None else _c.load_events(cfg)):
        if e.get("type") == "location_visited":
            return {"location": e.get("slug", ""), "area": e.get("area", "") or "",
                    "name": e.get("name") or e.get("slug", "")}
    return {}


def set_position(cfg: dict, **changes) -> dict:
    state = _c.load_state(cfg)
    pos = current_position(cfg, state)
    pos.update({k: v for k, v in changes.items() if k in _KEYS})
    h, m = state.get("current_hour", 0), state.get("current_minute", 0)
    pos["since"] = {"day": state.get("current_day", 1), "time": f"{h:02d}:{m:02d}"}
    state["position"] = pos
    _c.save_state(cfg, state)
    return pos


def position_label(pos: dict) -> str:
    if pos.get("label"):
        return f"{pos['label']} · {pos['map_name']}" if pos.get("map_name") else pos["label"]
    return pos.get("name") or pos.get("location") or ""
```

```python
class DungmlNotFound(DungmlUnavailable):
    """dungml answered 404 for something the caller named — an unknown room
    or door. `detail` carries dungml's body (candidates for a room)."""

    def __init__(self, message: str, detail=None):
        super().__init__(message)

class DungmlConflict(DungmlUnavailable):

def describe_error(e: DungmlUnavailable) -> dict:
    """A tool-result dict for a dungml failure the DM can act on."""
    if isinstance(e, DungmlConflict):
        return {"error": "the map is mid-edit in the dungml editor and does not "
                         "parse right now — narrate without it, try again later"}
    if isinstance(e, DungmlNotFound):

        out = {"error": d.get("error") or str(e)}
        if d.get("candidates"):
            out["candidates"] = d["candidates"]

    return {"error": str(e)}
```

```python
"""The party moves through one tool, and the room it enters comes back.

`move_party` updates both halves of state.json["position"] — the ttrpg2
location and the dungml room — moves the dungml party marker (which reveals
the room), writes one `location_visited` event and returns the room's
context from the authored map, split into `perceived` (what the characters
see) and `dm_only` (never narrated), plus this campaign's notes on it.

Room notes are what the party changed: "horse tied here", "cleared", "east
door barricaded". They live in room_notes.json keyed by map and node, so a
reset or a new session still knows.
"""
from __future__ import annotations

import json

import _campaign as _c
from tools import dungml_play as dp
from tools import position as _pos

_BRIEF_CHARS = 900


# ----- room notes -----

def room_key(map_id: str, node: str) -> str:
    return f"{map_id}:{node}"


def _notes_path(cfg: dict):
    return cfg["_dir"] / "room_notes.json"


def _load_notes(cfg: dict) -> dict:
    try:
        return json.loads(_notes_path(cfg).read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError):
        return {}


def notes_for(cfg: dict, map_id: str, node: str) -> list[dict]:
    return list(_load_notes(cfg).get(room_key(map_id, node), []))


def add_note(cfg: dict, map_id: str, node: str, text: str) -> dict:
    state = _c.load_state(cfg)
    note = {"day": state.get("current_day", 1),
            "time": f"{state.get('current_hour', 0):02d}:{state.get('current_minute', 0):02d}",
            "text": text.strip()}
    data = _load_notes(cfg)
    data.setdefault(room_key(map_id, node), []).append(note)
    _c.atomic_write_text(_notes_path(cfg), json.dumps(data, indent=2, ensure_ascii=False))
    return note


# ----- helpers -----

def _brief(path) -> str:
    text = path.read_text(encoding="utf-8").strip()
    return text if len(text) <= _BRIEF_CHARS else text[:_BRIEF_CHARS].rstrip() + "\n…"


def _campaign_map(cfg: dict) -> tuple[str | None, str | None]:
    link = dp.link_state(cfg)
    if link["source"] == "campaign" and link["map_id"]:
        return link["map_id"], link.get("map_name")
    return None, None


def register(mcp):

    @mcp.tool()
    def move_party(to: str = "", map: str = "", location: str = "") -> dict:
        """Move the party. The ONLY way the party changes place.

        to:       a room on the current map — its id (room.cave_1), bare name
                  (cave_1) or label ("39", "Grand Foyer"). Moves the map
                  marker and reveals the room.
        map:      switch to another authored map first (name substring, as
                  dungml_locales lists them) — stairs, a map exit.
        location: a ttrpg2 location slug (a town, the road, a dungeon as a
                  whole) — for moves with no map, or to change both at once.

        Returns the room just entered: `perceived` is what the characters
        see and may be narrated; `dm_only` never is. Also returns
        `room_notes` (what the party changed there earlier) and, when the
        location changed, the opening of its file. An unknown room returns
        `candidates` and moves nothing. If the map service is down the
        location half still moves and `warning` says so.
        """
        cfg = _c.load_campaign()
        ext = dp.external_id(cfg)
        changes: dict = {}
        out: dict = {}

        loc = None
        if location:
            loc = _pos.find_location(cfg, location)
            if loc is None:
                return {"error": f"no location file '{location}' — create_location first, "
                                 f"or pass `to` for a room on the map"}

        if map:
            switched = dp.use_locale(cfg, map)
            if "error" in switched:
                return switched

        if to:
            map_id, map_name = _campaign_map(cfg)
            try:
                if map_id:
                    res = dp.campaign_set_party(ext, map_id, to)
                else:
                    res = dp.set_party(ext, to)
                    map_id = dp.map_info(ext).get("map_id")
                    map_name = "combat map"
            except (dp.DungmlNotFound, dp.DungmlConflict) as e:
                return dp.describe_error(e)
            except dp.DungmlUnavailable as e:
                out["warning"] = f"map service unreachable — party marker not moved ({e})"
            else:
                room = res.get("room") or {}
                node = res.get("party_location") or room.get("id")
                changes.update(map_id=map_id, map_name=map_name, node=node,
                               label=room.get("label") or node)
                out["room"] = room
                out["room_notes"] = notes_for(cfg, map_id, node)
                dp.save_link_state(cfg, party_node=node)

        if loc is not None:
            changes.update(location=loc["slug"], area=loc["area"], name=loc["name"])
            out["location_brief"] = _brief(loc["path"])

        if not changes:
            if "warning" in out:
                return {"error": out["warning"]}
            return {"error": "nothing to move to — pass `to`, `map` or `location`"}

        pos = _pos.set_position(cfg, **changes)
        _c.append_event(cfg, {
            "type": "location_visited",
            "slug": pos.get("location", ""), "area": pos.get("area", ""),
            "name": _pos.position_label(pos),
            "map_id": pos.get("map_id"), "node": pos.get("node"), "label": pos.get("label"),
        })
        out["position"] = pos
        return out

    @mcp.tool()
    def room(query: str = "") -> dict:
        """The authored room: the current one, or any room on the current map
        by id, bare name or label. `perceived` may be narrated; `dm_only`
        never. Read-only — for "what's through that door", or re-reading
        the room after a reset. Includes the campaign's notes on it."""
        cfg = _c.load_campaign()
        map_id, _ = _campaign_map(cfg)
        if not map_id:
            return {"error": "room context needs an authored locale (dungml_use_locale)"}
        pos = _pos.current_position(cfg)
        q = query or pos.get("node")
        if not q:
            return {"error": "the party isn't in a room yet — move_party first"}
        try:
            ctx = dp.campaign_room(dp.external_id(cfg), map_id, q)
        except dp.DungmlUnavailable as e:
            return dp.describe_error(e)
        return {"room": ctx, "room_notes": notes_for(cfg, map_id, ctx.get("id", q))}

    @mcp.tool()
    def door(door: str, state: str = "", found: bool = False) -> dict:
        """Record what happened to a door: opened, forced, locked, barred —
        or a secret door found (found=True).

        door: a door key ("14,9", from a room's exits) or the room it leads
              to from the current room ("37", "room.cave_2")."""
        cfg = _c.load_campaign()
        map_id, _ = _campaign_map(cfg)
        if not map_id:
            return {"error": "doors are tracked on an authored locale (dungml_use_locale)"}
        is_key = "," in door
        between = None
        if not is_key:
            here = _pos.current_position(cfg).get("node")
            if not here:
                return {"error": "the party isn't in a room yet — move_party first"}
            between = [here, door]
        try:
            return dp.campaign_door(dp.external_id(cfg), map_id,
                                    door=door if is_key else None, between=between,
                                    discovered=True, state=state or None)
        except dp.DungmlUnavailable as e:
            return dp.describe_error(e)

    @mcp.tool()
    def note_room(text: str, query: str = "") -> dict:
        """Remember what the party changed in a room — cleared it, left the
        horse, barricaded a door, looted the chest. Returned by move_party
        and room() every time the party is there again. Current room by
        default; `query` names another by id or label."""
        cfg = _c.load_campaign()
        pos = _pos.current_position(cfg)
        map_id = pos.get("map_id")
        node = pos.get("node")
        if query:
            mid, _ = _campaign_map(cfg)
            if not mid:
                return {"error": "naming a room needs an authored locale"}
            try:
                node = dp.campaign_room(dp.external_id(cfg), mid, query, scope="fog")["id"]
            except dp.DungmlUnavailable as e:
                return dp.describe_error(e)
            map_id = mid
        if not (map_id and node):
            return {"error": "the party isn't in a room yet — move_party first"}
        return {"room": node, "note": add_note(cfg, map_id, node, text)}
```

```python
"""module_lookup — search the published module the campaign is running.

Campaigns that run a module keep its text under campaigns/<slug>/reference/
as markdown. The DM used to read those files with Bash (or not at all, and
improvise); this indexes them into an in-memory SQLite FTS5 table, one row
per paragraph or numbered key entry, tagged with its heading path. The index
is rebuilt when any reference file's mtime changes.

DM-only material: module text is never narrated verbatim and never leaks
into player-facing choices (CLAUDE.md, Hard Constraint #5).
"""
from __future__ import annotations

import re
import sqlite3

import _campaign as _c

_index: dict[str, tuple[tuple, sqlite3.Connection]] = {}
_HEADING = re.compile(r"^(#{1,6})\s+(.+?)\s*$")
_BOLD_HEADING = re.compile(r"^\*\*(.+?)\*\*\s*$")
_KEY_ENTRY = re.compile(r"^\d+[a-z]?\.\s")


def _chunks(text: str) -> list[tuple[str, str]]:
    """(heading path, chunk) pairs. A chunk is a paragraph, or one numbered
    key entry with its wrapped continuation lines."""
    path: list[tuple[int, str]] = []
    out: list[tuple[str, str]] = []
    buf: list[str] = []

    def flush():
        if buf:
            out.append((" › ".join(t for _, t in path), " ".join(buf).strip()))
            buf.clear()

    for raw in text.splitlines():
        line = raw.strip()
        m = _HEADING.match(line)
        b = _BOLD_HEADING.match(line)
        if m or b:
            flush()
            level = len(m.group(1)) if m else 7
            title = (m.group(2) if m else b.group(1)).strip()
            while path and path[-1][0] >= level:
                path.pop()
            path.append((level, title))
            continue
        if not line:
            flush()
            continue
        if _KEY_ENTRY.match(line):
            flush()
        buf.append(line)
    flush()
    return out


def _db(cfg: dict) -> sqlite3.Connection | None:
    ref = cfg["_dir"] / "reference"
    files = sorted(ref.glob("*.md")) if ref.is_dir() else []
    if not files:
        return None
    sig = tuple((f.name, f.stat().st_mtime) for f in files)
    cached = _index.get(cfg["_name"])
    if cached and cached[0] == sig:
        return cached[1]
    con = sqlite3.connect(":memory:", check_same_thread=False)
    con.execute("CREATE VIRTUAL TABLE chunks USING fts5(file, path, text)")
    for f in files:
        rows = [(f.name, p, t) for p, t in _chunks(f.read_text(encoding="utf-8"))]
        con.executemany("INSERT INTO chunks VALUES (?, ?, ?)", rows)
    _index[cfg["_name"]] = (sig, con)
    return con


def _fts_query(q: str) -> str:
    """Each word as a quoted term, ANDed — user text is never FTS syntax."""
    words = re.findall(r"[\w'-]+", q)
    return " ".join('"' + w.replace('"', "") + '"' for w in words)


def register(mcp):

    @mcp.tool()
    def module_lookup(query: str, section: str = "", limit: int = 5) -> dict:
        """Search the published module this campaign runs (its reference/
        text): room keys, level overviews, NPCs, wandering-monster tables.

        query:   words to find, e.g. "fire beetles", "gentlemen ghouls".
        section: only results whose heading path contains this, e.g.
                 "1B", "Level Two", "Room Key".
        DM-only. The dungml map already carries most room keys as each
        room's boxed text and DM notes (move_party / room return them);
        reach for this for what the map doesn't hold."""
        cfg = _c.load_campaign()
        con = _db(cfg)
        if con is None:
            return {"results": [], "note": "this campaign has no reference/ module text"}
        fts = _fts_query(query)
        if not fts:
            return {"results": []}
        rows = con.execute(
            "SELECT file, path, text FROM chunks WHERE chunks MATCH ? ORDER BY rank LIMIT ?",
            (fts, max(1, int(limit)) * 4)).fetchall()
        if section:
            low = section.lower()
            rows = [r for r in rows if low in r[1].lower()]
        return {"results": [{"file": f, "path": p, "text": t}
                            for f, p, t in rows[:max(1, int(limit))]]}
```
